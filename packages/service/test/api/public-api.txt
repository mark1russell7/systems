now), 'cap'}\n    end\n  end\nend\nlocal remaining\nlocal state\nif kind == 'token-bucket' then\n  local span = a * b\n  local base = now\n  if full and full > now then base = full end\n  local nxt = base + a * units\n  if nxt -
